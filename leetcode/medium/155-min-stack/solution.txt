class MinStack {
    Stack<int[]> st;
    //0th index is main stack element 
    //1th index is min so far

    public MinStack() {
        st = new Stack<>();
    }

    public void push(int value) {
        if(st.isEmpty()){
            st.push(new int[]{value,value});
            return;
        }
        int[] arr = st.peek();
        int minSoFar = arr[1];

            st.push(new int[]{value,minSoFar});
