 < n; i++) {

            sum += nums[i];

            char[] arr = String.valueOf(nums[i]).toCharArray();

            Arrays.sort(arr);

            int max = 0;

            for (int j = arr.length - 1; j >= 0; j--) {
                max = max * 10 + (arr[j] - '0');
            }

            gains[i] = max - nums[i];
        }

        Arrays.sort(gains);

        for (int i = n - 1; i >= 0 && k > 0; i--) {
            sum += gains[i];
            k--;
        }

        return sum;

    }
