class Solution {
    public int maxArea(int[] height) {
        int ans = 0;
        int j = height.length-1;
        int i = 0;

        while(i<j){
            int min = Math.min(height[i],height[j]);
            int prod = min * (j-i);
            ans = Math.max(prod,ans);
            if(height[i]>height[j]){
                j--;
            }else{
                i++;
            }

        }

        return ans;
        
    }
}
