class Solution {
    public int maximumProduct(int[] nums, int k) {
        PriorityQueue<Integer> pq = new PriorityQueue<>();

        for(int ele: nums){
            pq.offer(ele);
        }

        while(k>0){
            int ele = pq.poll();
            ele++;
            k--;
            pq.offer(ele);
        }

      
        long ans = 1;
        int mod = 1_000_000_007;

        while (!pq.isEmpty()) {
            ans = (ans * pq.poll()) % mod;
        }

    return (int) ans;
        
    }
}
