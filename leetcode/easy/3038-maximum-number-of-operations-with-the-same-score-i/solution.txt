class Solution {
    public int maxOperations(int[] nums) {
        int maxScore = nums[0]+nums[1];
        int count = 1;
        int i = 2;
        while(i<nums.length-1){
            if(nums[i]+nums[i+1]==maxScore){
                count++;
                i+=2;
            }else{
                break;
            }
        }
        return count;
    }
}
